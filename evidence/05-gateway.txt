$ compose exec -T gateway cat /etc/nginx/conf.d/default.conf
# Gateway for the Lost & Found API.
#
# Goes in the http{} block (conf.d/api.conf or sites-enabled/api.conf).
#
# The app must be started with the prefix it is served under, or every
# redirect and every URL in the OpenAPI schema will be missing /api/v1:
#
#     uvicorn main:app --host 0.0.0.0 --port 8000 --root-path /api/v1
#
# (or FastAPI(title=..., root_path="/api/v1") in main.py)

upstream app_pool {
    least_conn;
    server app:8000 max_fails=3 fail_timeout=10s;
    keepalive 32;
}

log_format api '$remote_addr $request_id "$request" $status '
               'upstream=$upstream_addr rt=$request_time urt=$upstream_response_time';

server {
    listen 80;
    server_name _;

    access_log /var/log/nginx/api.access.log api;

    client_max_body_size 1m;

    default_type application/json;

    proxy_http_version 1.1;
    proxy_set_header Connection        "";

    proxy_set_header Host              $host;
    proxy_set_header X-Real-IP         $remote_addr;
    proxy_set_header X-Forwarded-For   $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;
    proxy_set_header X-Forwarded-Prefix /api/v1;
    proxy_set_header X-Request-ID      $request_id;

    proxy_connect_timeout 3s;
    proxy_send_timeout    30s;
    proxy_read_timeout    30s;

    proxy_intercept_errors off;

    add_header X-Request-ID $request_id always;

    # Routes

    location = /healthz {
        default_type text/plain;
        return 200 "Gateway OK \n";
    }

    location = /readyz {
        proxy_pass http://app_pool/health;
    }

    location = /api/v1 {
        return 308 /api/v1/;
    }

    location = /docs {
        proxy_pass http://app_pool/docs;
    }

    location = /openapi.json {
        proxy_pass http://app_pool/openapi.json;
    }

    location /api/v1/ {
        proxy_pass http://app_pool/;
    }

    location / {
        return 404 '{"error":{"code":"NOT_FOUND","detail":"no route at the gateway","fields":null,"request_id":"$request_id"}}';
    }

    error_page 502 503 504 = @upstream_down;

    location @upstream_down {
        return 503 '{"error":{"code":"SERVICE_UNAVAILABLE","detail":"the service is temporarily unavailable, please try again","fields":null,"request_id":"$request_id"}}';
    }
}
